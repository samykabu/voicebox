#!/usr/bin/env bash
# Add one dependency at a time until `$CHECK` succeeds. Each install uses the project constraints.
W=/c/Users/sabus/voxcpm-nodeps/Scripts
C=/c/Users/sabus/voxcpm-nodeps/project-constraints.txt
CHECK="$1"          # python snippet to run
MAXIT=${2:-25}
declare -A MAP=( [yaml]=pyyaml [sklearn]=scikit-learn [PIL]=pillow [huggingface_hub]=huggingface-hub [google]=protobuf [cv2]=opencv-python )
for i in $(seq 1 $MAXIT); do
  echo "---- iteration $i  $(date -Iseconds)"
  OUT=$($W/python.exe -c "$CHECK" 2>&1); RC=$?
  echo "$OUT" | tail -6
  if [ $RC -eq 0 ]; then echo "CHECK PASSED at iteration $i"; exit 0; fi
  MOD=$(echo "$OUT" | grep -oE "No module named '[^']+'" | tail -1 | sed -E "s/No module named '([^'.]+).*'/\1/")
  if [ -z "$MOD" ]; then echo "STOP: failure is not a missing module; needs manual action"; exit 2; fi
  PKG=${MAP[$MOD]:-$MOD}
  echo ">> missing module '$MOD' -> pip install $PKG  (-c project-constraints)"
  S=$(date +%s)
  $W/python.exe -m pip install "$PKG" -c "$C" 2>&1 | grep -E "Successfully installed|ERROR|error|conflict|requires" | head -20
  echo ">> install dur=$(( $(date +%s)-S ))s"
done
echo "STOP: max iterations"; exit 3
