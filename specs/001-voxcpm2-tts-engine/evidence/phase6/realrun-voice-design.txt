"""Phase 6 real voice-design run (T035 / quickstart Step 6): resolve helper + VoxCPMBackend on CUDA."""

import asyncio
import hashlib
import os
import sys
import time
from datetime import datetime
from pathlib import Path

# The model is already in the default HF cache; prove loading needs no network.
os.environ["HF_HUB_OFFLINE"] = "1"
os.environ["TRANSFORMERS_OFFLINE"] = "1"

sys.path.insert(0, r"D:\Projects\Personal\voicebox")

import numpy as np  # noqa: E402
import soundfile as sf  # noqa: E402
import torch  # noqa: E402

from backend import config  # noqa: E402

OUT = Path(r"C:\Users\sabus\voxcpm-probe-out\phase6")
OUT.mkdir(parents=True, exist_ok=True)
config.set_data_dir(OUT / "data")  # keep any cache writes outside the repo

from backend.backends.voxcpm_backend import VoxCPMBackend  # noqa: E402
from backend.services.generation import resolve_backend_instruct  # noqa: E402
from backend.utils.audio import save_audio  # noqa: E402

ENGINE = "voxcpm"
DELIVERY = "whisper softly"
DESCRIPTION = "An old man, deep, slow and raspy voice"
SEED = 42
CASES = [
    ("design_en_seed42", "Hello, this voice was designed from a written description."),
    ("design_ar_seed42", "مرحبا، هذا صوت مصمم من وصف مكتوب."),
]


def sha(path: Path) -> str:
    return hashlib.sha256(path.read_bytes()).hexdigest()


async def main() -> None:
    print(f"# {datetime.now().isoformat(timespec='seconds')}")
    print(f"torch={torch.__version__} cuda_available={torch.cuda.is_available()} device0={torch.cuda.get_device_name(0)}")
    print(f"HF_HUB_OFFLINE={os.environ['HF_HUB_OFFLINE']}")

    resolved = resolve_backend_instruct(ENGINE, DELIVERY, DESCRIPTION)
    print(f"resolve_backend_instruct({ENGINE!r}, instruct={DELIVERY!r}, voice_description={DESCRIPTION!r}) -> {resolved!r}")
    print(f"resolve_backend_instruct({ENGINE!r}, instruct={DELIVERY!r}, voice_description=None) -> "
          f"{resolve_backend_instruct(ENGINE, DELIVERY, None)!r}")
    assert resolved == DESCRIPTION

    b = VoxCPMBackend()
    torch.cuda.reset_peak_memory_stats()
    t0 = time.perf_counter()
    await b.load_model()
    load_s = time.perf_counter() - t0
    print(f"device={b.device} load_s={load_s:.2f} output_rate={int(b.model.tts_model.sample_rate)}")

    captured: list[dict] = []
    real_generate = b.model.generate

    def capturing_generate(**kwargs):
        captured.append(dict(kwargs))
        return real_generate(**kwargs)

    b.model.generate = capturing_generate

    for label, text in CASES:
        t0 = time.perf_counter()
        audio, sr = await b.generate(text, {}, language="en", seed=SEED, instruct=resolved)
        wall = time.perf_counter() - t0
        path = OUT / f"{label}.wav"
        save_audio(audio, str(path), sr)
        info = sf.info(str(path))
        duration = audio.shape[0] / sr
        vendor = captured[-1]
        vt = vendor["text"]
        expected = f"({DESCRIPTION}){text}"
        print(
            f"{label}: wall_s={wall:.2f} samples={audio.shape[0]} returned_sr={sr} file_sr={info.samplerate} "
            f"duration_s={duration:.3f} rtf={wall / duration:.3f} rms={float(np.sqrt(np.mean(audio**2))):.4f} "
            f"file={path} sha256={sha(path)}"
        )
        print(f"  vendor text: {vt!r}")
        print(f"  vendor text utf-8 hex: {vt.encode('utf-8').hex()}")
        print(f"  vendor text == '(description)' + text byte for byte: {vt.encode('utf-8') == expected.encode('utf-8')}")
        print(f"  delivery text present: {'whisper' in vt}; full-width brackets present: {'（' in vt or '）' in vt}")
        print(f"  prompt_wav_path={vendor['prompt_wav_path']!r} prompt_text={vendor['prompt_text']!r} "
              f"cfg_value={vendor['cfg_value']} inference_timesteps={vendor['inference_timesteps']} normalize={vendor['normalize']}")

    print(f"peak_cuda_allocated_mb={torch.cuda.max_memory_allocated() / 2**20:.0f} "
          f"peak_cuda_reserved_mb={torch.cuda.max_memory_reserved() / 2**20:.0f}")
    b.unload_model()
    print(f"unloaded is_loaded={b.is_loaded()}")
    print("NOTE: whether the voice matches the description (old man, deep, slow, raspy) needs a human listen.")


if __name__ == "__main__":
    asyncio.run(main())
