"""T051 real run: one short VoxCPM2 designed-voice utterance saved through save_audio, tag read back."""

import asyncio
import os
import struct
import sys
import time
from datetime import datetime
from pathlib import Path

os.environ["HF_HUB_OFFLINE"] = "1"
os.environ["TRANSFORMERS_OFFLINE"] = "1"

sys.path.insert(0, r"D:\Projects\Personal\voicebox")

import soundfile as sf  # noqa: E402
import torch  # noqa: E402

from backend import config  # noqa: E402

OUT = Path(r"C:\Users\sabus\voxcpm-probe-out\phase6")
OUT.mkdir(parents=True, exist_ok=True)
config.set_data_dir(OUT / "data")

from backend.backends.voxcpm_backend import VoxCPMBackend  # noqa: E402
from backend.services.generation import resolve_backend_instruct  # noqa: E402
from backend.utils.audio import AI_GENERATED_DISCLOSURE, save_audio  # noqa: E402

TARGET = OUT / "t051_tagged.wav"
TEXT = "This audio was generated by Voicebox."
DESCRIPTION = "A young woman, gentle and sweet voice"


def riff_chunks(raw: bytes):
    pos, out = 12, []
    while pos + 8 <= len(raw):
        cid = raw[pos : pos + 4]
        (size,) = struct.unpack("<I", raw[pos + 4 : pos + 8])
        out.append((cid, pos, size, raw[pos + 8 : pos + 8 + size]))
        pos += 8 + size + (size & 1)
    return out


async def main() -> None:
    print(f"# {datetime.now().isoformat(timespec='seconds')}")
    print(f"torch={torch.__version__} cuda_available={torch.cuda.is_available()} sf={sf.__version__} libsndfile={sf.__libsndfile_version__}")
    b = VoxCPMBackend()
    t0 = time.perf_counter()
    await b.load_model()
    print(f"device={b.device} load_s={time.perf_counter() - t0:.2f}")
    instruct = resolve_backend_instruct("voxcpm", None, DESCRIPTION)
    t0 = time.perf_counter()
    audio, sr = await b.generate(TEXT, {}, seed=42, instruct=instruct)
    wall = time.perf_counter() - t0
    save_audio(audio, str(TARGET), sr)
    print(f"generated: wall_s={wall:.2f} samples={audio.shape[0]} returned_sr={sr} duration_s={audio.shape[0] / sr:.3f}")
    print(f"saved via backend.utils.audio.save_audio -> {TARGET} ({TARGET.stat().st_size} bytes); tmp left: {Path(str(TARGET) + '.tmp').exists()}")

    info = sf.info(str(TARGET))
    print(f"sf.info: samplerate={info.samplerate} channels={info.channels} frames={info.frames} duration_s={info.duration:.3f} format={info.format} subtype={info.subtype}")
    with sf.SoundFile(str(TARGET)) as f:
        print(f"soundfile read-back: comment={f.comment!r} software={f.software!r} copy_metadata={f.copy_metadata()!r}")
        assert f.comment == AI_GENERATED_DISCLOSURE

    raw = TARGET.read_bytes()
    print(f"RIFF header: {raw[:4]!r} size={struct.unpack('<I', raw[4:8])[0]} form={raw[8:12]!r}")
    for cid, pos, size, body in riff_chunks(raw):
        if cid == b"LIST":
            print(f"chunk {cid!r} @ offset {pos} size {size}: raw bytes = {body!r}")
            p = 4
            while p + 8 <= len(body):
                tid = body[p : p + 4]
                (tsz,) = struct.unpack("<I", body[p + 4 : p + 8])
                print(f"  INFO {tid.decode()}: {body[p + 8 : p + 8 + tsz].rstrip(bytes(1)).decode('utf-8')!r}")
                p += 8 + tsz + (tsz & 1)
        else:
            print(f"chunk {cid!r} @ offset {pos} size {size}")
    b.unload_model()


if __name__ == "__main__":
    asyncio.run(main())
